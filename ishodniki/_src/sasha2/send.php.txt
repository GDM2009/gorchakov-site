<?php
// Форма обратной связи сайта {{SITE}}: письмо эксперту и резервная копия в _data/obrascheniya.csv.
// Файл создаёт сборщик из _src/sasha2/send.php.txt; адрес получателя берётся из CONTACTS в build.js.
date_default_timezone_set('Europe/Moscow');
header('Content-Type: text/plain; charset=utf-8');
header('Cache-Control: no-store');
header('X-Robots-Tag: noindex');
if ($_SERVER['REQUEST_METHOD'] !== 'POST') { http_response_code(405); exit('method'); }

const MAIL_TO = '{{MAIL_TO}}';
const MAIL_FROM = 'no-reply@{{HOST}}';

$clean = function ($v, $max) {
  if (is_array($v)) return '';
  $s = preg_replace('/[\x00-\x08\x0B\x0C\x0E-\x1F\x7F]/u', '', (string)$v);
  return mb_substr(trim((string)$s), 0, $max);
};
$labels = ['name' => 'Имя', 'contact' => 'Телефон или почта', 'topic' => 'Тема', 'message' => 'Сообщение', 'page' => 'Страница'];
$d = [];
foreach ($labels as $k => $_) $d[$k] = $clean($_POST[$k] ?? '', $k === 'message' ? 4000 : 300);
// поле-ловушка: человек его не видит, а робот заполняет
if ($clean($_POST['website'] ?? '', 100) !== '') { echo 'ok'; exit; }
if ($d['name'] === '' || $d['contact'] === '') { http_response_code(400); exit('empty'); }
if (preg_match('/[\r\n]/', $d['name'] . $d['contact'] . $d['topic'])) { http_response_code(400); exit('bad'); }

// не больше 5 обращений с одного адреса за 10 минут
$dir = __DIR__ . '/_data';
if (!is_dir($dir)) @mkdir($dir, 0750, true);
$ip = $_SERVER['HTTP_X_REAL_IP'] ?? ($_SERVER['REMOTE_ADDR'] ?? '');
$rl = $dir . '/rl_' . md5($ip) . '.txt'; $now = time();
$hits = is_file($rl) ? array_filter(array_map('intval', file($rl) ?: []), fn($t) => $t > $now - 600) : [];
if (count($hits) >= 5) { http_response_code(429); exit('limit'); }
$hits[] = $now; @file_put_contents($rl, implode("\n", $hits), LOCK_EX);

// резервная копия на случай, если письмо не дойдёт
$file = $dir . '/obrascheniya.csv'; $new = !file_exists($file);
if ($fh = @fopen($file, 'a')) {
  flock($fh, LOCK_EX);
  if ($new) { fwrite($fh, "\xEF\xBB\xBF"); fputcsv($fh, array_merge(['Дата'], array_values($labels))); }
  fputcsv($fh, array_map(fn($v) => preg_replace('/^[=+\-@]/', "'$0", $v), array_merge([date('d.m.Y H:i')], array_values($d))));
  flock($fh, LOCK_UN); fclose($fh);
}

$body = "Обращение с сайта {{SITE}}\n\n";
foreach ($labels as $k => $l) if ($d[$k] !== '') $body .= $l . ': ' . $d[$k] . "\n";
$subj = 'Обращение с сайта: ' . ($d['topic'] !== '' ? $d['topic'] . ' — ' : '') . $d['name'];
$from = '=?UTF-8?B?' . base64_encode('Сайт Александра Горчакова') . '?=';
$headers = "From: " . $from . " <" . MAIL_FROM . ">\r\n"
  . (filter_var($d['contact'], FILTER_VALIDATE_EMAIL) ? "Reply-To: " . $d['contact'] . "\r\n" : '')
  . "MIME-Version: 1.0\r\nContent-Type: text/plain; charset=utf-8\r\nContent-Transfer-Encoding: 8bit";
$ok = @mail(MAIL_TO, '=?UTF-8?B?' . base64_encode($subj) . '?=', $body, $headers, '-f' . MAIL_FROM);
if (!$ok) { @file_put_contents($dir . '/mail_failed.log', date('c') . ' ' . $d['name'] . "\n", FILE_APPEND | LOCK_EX); http_response_code(502); exit('mail'); }
echo 'ok';
